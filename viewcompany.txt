import axios from "axios";
import { useState } from "react";
import { useCompany } from "../../context/CompanyContext";
import { useAuth } from "../../context/AuthContext";

const ViewCompany = () => {
  const { company } = useCompany();
  const { userEmail } = useAuth();

  const [activeTab, setActiveTab] = useState("job");
  const [isApplying, setIsApplying] = useState(false);

  // Popup states
  const [showPopup, setShowPopup] = useState(false);
  const [resumeLink, setResumeLink] = useState("");
  const [isChecked, setIsChecked] = useState(false);

  // Optional local flag so the button reflects success immediately
  const [appliedLocal, setAppliedLocal] = useState(false);

  // Check if current user has already applied
  const hasApplied =
    appliedLocal || company?.applied_students?.includes(userEmail);

  const handleApply = async () => {
    if (!userEmail || !company) return;

    setIsApplying(true);
    try {
      const response = await axios.put(
        `http://localhost:5000/company/${company._id}`,
        {
          email: userEmail,
          resumeLink: resumeLink,
        }
      );

      if (response.status === 200) {
        // Close popup and reflect UI state
        setShowPopup(false);
        setAppliedLocal(true);
      }
    } catch (error) {
      console.error("Error applying to company:", error);
    } finally {
      setIsApplying(false);
    }
  };

  if (!company) {
    return (
      <div className="font-instrument min-h-screen py-8 px-4 w-full max-w-[75rem] mx-auto flex items-center justify-center">
        <div className="text-center">
          <div className="animate-spin rounded-full h-12 w-12 border-b-2 border-[#72265F] mx-auto mb-4"></div>
          <p className="text-gray-600">Loading company details...</p>
        </div>
      </div>
    );
  }

  return (
    <div className="font-instrument min-h-screen py-8 px-4 w-full max-w-[75rem] mx-auto flex flex-col">
      {/* Header */}
      <div className="mb-6">
        <h1 className="text-3xl font-bold text-[#72265F] text-center">Company Details</h1>
      </div>

      {/* Main Content Container */}
      <div className="bg-white rounded-2xl shadow-lg border border-gray-200 flex-1">
        {/* Company Header Section */}
        <div className="px-8 py-6 border-b border-gray-200 rounded-t-2xl bg-gray-50">
          <div className="flex items-center justify-between">
            {/* Left: Logo and Company Info */}
            <div className="flex items-center">
              <div className="flex-shrink-0 w-20 h-20 rounded-xl overflow-hidden border-2 border-gray-200 bg-gray-50 p-2">
                <img
                  src={company.logo || "https://upload.wikimedia.org/wikipedia/en/4/45/Ciena_logo.svg"}
                  alt={`${company.title || "Company"} logo`}
                  className="w-full h-full object-contain"
                  onError={(e) => {
                    e.target.src = "https://upload.wikimedia.org/wikipedia/en/4/45/Ciena_logo.svg";
                  }}
                />
              </div>

              <div className="ml-6">
                <h2 className="text-2xl font-bold text-[#72265F] mb-1">
                  {company.title || "Unknown Company"}
                </h2>
                <h3 className="text-xl font-semibold text-gray-700 mb-2">
                  {company.role || "N/A"}
                </h3>
                <div className="flex items-center gap-4 text-sm text-gray-600">
                  <span className="inline-flex items-center px-3 py-1 rounded-full bg-gray-100 text-gray-800">
                    {company.location || "N/A"}
                  </span>
                  <span className="inline-flex items-center px-3 py-1 rounded-full bg-gray-200 text-gray-900">
                    {company.job_type || "N/A"}
                  </span>
                  <span className="font-semibold text-black">
                    CTC: {company.ctc || "N/A"}
                  </span>
                </div>
              </div>
            </div>

            {/* Right: Apply Button */}
            <div className="ml-6">
              {hasApplied ? (
                <button
                  disabled
                  className="px-6 py-3 bg-green-100 text-green-700 border border-green-300 rounded-full font-semibold cursor-not-allowed"
                >
                  Applied ✓
                </button>
              ) : (
                <button
                  onClick={() => setShowPopup(true)}
                  disabled={isApplying || !userEmail}
                  className="bg-[#72265F] text-white px-6 py-3 rounded-full text-sm font-semibold hover:bg-[#913e7c] transition-all duration-300 shadow-md hover:shadow-lg transform hover:scale-105 active:scale-95 disabled:bg-gray-400 disabled:cursor-not-allowed disabled:transform-none"
                >
                  {isApplying ? "Applying..." : "Apply Now"}
                </button>
              )}
            </div>
          </div>
        </div>

        {/* Tab Navigation */}
        <div className="px-8 py-4 border-b border-gray-200">
          <div className="flex bg-white rounded-full p-1 shadow-sm border border-gray-200 max-w-2xl">
            <button
              onClick={() => setActiveTab("job")}
              className={`px-6 py-3 rounded-full font-semibold transition-all duration-300 flex-1 ${
                activeTab === "job" 
                  ? "bg-[#72265F] text-white shadow-sm" 
                  : "text-[#72265F] hover:bg-gray-50"
              }`}
            >
              Job Description
            </button>
            <button
              onClick={() => setActiveTab("workflow")}
              className={`px-6 py-3 rounded-full font-semibold transition-all duration-300 flex-1 ${
                activeTab === "workflow" 
                  ? "bg-[#72265F] text-white shadow-sm" 
                  : "text-[#72265F] hover:bg-gray-50"
              }`}
            >
              Hiring Workflow
            </button>
            <button
              onClick={() => setActiveTab("eligibility")}
              className={`px-6 py-3 rounded-full font-semibold transition-all duration-300 flex-1 ${
                activeTab === "eligibility" 
                  ? "bg-[#72265F] text-white shadow-sm" 
                  : "text-[#72265F] hover:bg-gray-50"
              }`}
            >
              Eligibility
            </button>
          </div>
        </div>

        {/* Tab Content */}
        <div className="px-8 py-6 h-[60vh] overflow-y-auto custom-scrollbar">
          {activeTab === "job" && (
            <div className="space-y-6">
              {/* Job Description */}
              <div className="bg-gray-50 rounded-xl p-6 border border-gray-100">
                <h3 className="text-xl font-bold text-[#72265F] mb-4">Job Description</h3>
                <p className="text-gray-700 leading-relaxed">
                  {company.description || "No job description provided."}
                </p>
              </div>

              {/* Required Skills */}
              <div className="bg-gray-50 rounded-xl p-6 border border-gray-100">
                <h3 className="text-xl font-bold text-[#72265F] mb-4">Required Skills</h3>
                {company.required_skills && company.required_skills.length > 0 ? (
                  <div className="flex flex-wrap gap-2">
                    {company.required_skills.map((skill, i) => (
                      <span key={i} className="inline-flex items-center px-3 py-1 rounded-full text-sm font-medium bg-white text-gray-800 border border-gray-300">
                        {skill}
                      </span>
                    ))}
                  </div>
                ) : (
                  <p className="text-gray-500">No specific skills mentioned.</p>
                )}
              </div>

              {/* Additional Information */}
              <div className="bg-gray-50 rounded-xl p-6 border border-gray-100">
                <h3 className="text-xl font-bold text-[#72265F] mb-4">Additional Information</h3>
                <p className="text-gray-700 leading-relaxed">
                  {company.additional_info || "No additional information provided."}
                </p>
              </div>

              {/* Job Function & Profile */}
              {(company.job_function || company.job_profile) && (
                <div className="grid md:grid-cols-2 gap-4">
                  {company.job_function && (
                    <div className="bg-gray-50 rounded-xl p-6 border border-gray-100">
                      <h3 className="text-lg font-bold text-[#72265F] mb-2">Job Function</h3>
                      <p className="text-gray-700">{company.job_function}</p>
                    </div>
                  )}
                  {company.job_profile && (
                    <div className="bg-gray-50 rounded-xl p-6 border border-gray-100">
                      <h3 className="text-lg font-bold text-[#72265F] mb-2">Job Profile</h3>
                      <p className="text-gray-700">{company.job_profile}</p>
                    </div>
                  )}
                </div>
              )}
            </div>
          )}

          {activeTab === "workflow" && (
            <div className="bg-gray-50 rounded-xl p-6 border border-gray-100">
              <h3 className="text-xl font-bold text-[#72265F] mb-4">Hiring Workflow</h3>
              <div className="prose prose-sm max-w-none">
                <p className="text-gray-700 leading-relaxed whitespace-pre-line">
                  {company.hiring_workflow || "No hiring workflow specified."}
                </p>
              </div>
            </div>
          )}

          {activeTab === "eligibility" && (
            <div className="space-y-6">
              {/* Eligibility Criteria */}
              <div className="bg-gray-50 rounded-xl p-6 border border-gray-100">
                <h3 className="text-xl font-bold text-[#72265F] mb-4">Eligibility Criteria</h3>
                <p className="text-gray-700 leading-relaxed">
                  {company.eligibility || "No specific eligibility criteria mentioned."}
                </p>
              </div>

              {/* Applicable Courses */}
              <div className="bg-gray-50 rounded-xl p-6 border border-gray-100">
                <h3 className="text-xl font-bold text-[#72265F] mb-4">Applicable Courses</h3>
                {company.applicable_courses && company.applicable_courses.length > 0 ? (
                  <div className="flex flex-wrap gap-2">
                    {company.applicable_courses.map((course, i) => (
                      <span key={i} className="inline-flex items-center px-3 py-1 rounded-full text-sm font-medium bg-white text-gray-800 border border-gray-300">
                        {course}
                      </span>
                    ))}
                  </div>
                ) : (
                  <p className="text-gray-500">No specific courses mentioned.</p>
                )}
              </div>
            </div>
          )}
        </div>

        {/* Job Description Link */}
        {company.JD && (
          <div className="px-8 py-4 border-t border-gray-200 bg-gray-50 rounded-b-2xl">
            <a
              href={company.JD}
              target="_blank"
              rel="noopener noreferrer"
              className="inline-flex items-center text-[#72265F] font-medium hover:text-[#913e7c] transition-colors"
            >
              View Detailed Job Description
              <svg className="w-4 h-4 ml-1" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M10 6H6a2 2 0 00-2 2v10a2 2 0 002 2h10a2 2 0 002-2v-4M14 4h6m0 0v6m0-6L10 14" />
              </svg>
            </a>
          </div>
        )}
      </div>

      {/* Application Popup Modal */}
      {showPopup && (
        <div className="fixed inset-0 bg-black bg-opacity-50 flex justify-center items-center z-50">
          <div className="bg-white p-8 rounded-2xl max-w-2xl w-full mx-4 shadow-2xl">
            <h2 className="text-2xl font-bold text-[#72265F] mb-6">Submit Application</h2>

            {/* Resume Link Input */}
            <div className="mb-6">
              <label className="block text-sm font-semibold text-gray-700 mb-2">
                Resume Link *
              </label>
              <input
                type="url"
                value={resumeLink}
                onChange={(e) => setResumeLink(e.target.value)}
                placeholder="Enter your resume link (Google Drive, OneDrive, etc.)"
                className="w-full border border-gray-300 rounded-lg p-3 focus:ring-2 focus:ring-[#72265F] focus:border-transparent transition-all"
              />
            </div>

            {/* Declaration */}
            <div className="mb-6 p-4 bg-gray-50 rounded-lg border border-gray-200">
              <div className="flex items-start gap-3">
                <input
                  type="checkbox"
                  checked={isChecked}
                  onChange={(e) => setIsChecked(e.target.checked)}
                  className="mt-1 w-4 h-4 text-[#72265F] border-gray-300 rounded focus:ring-[#72265F]"
                />
                <div className="text-sm text-gray-700 leading-relaxed">
                  <p className="font-semibold mb-2">Declaration:</p>
                  <p className="mb-2">
                    I hereby declare that I will make myself available for all
                    mandatory events related to the Placement Drive, including but
                    not limited to the Pre-Placement Talk, Online Assessment, and
                    Interview Rounds, as per the communicated schedule.
                  </p>
                  <p className="mb-2">
                    In case of an unavoidable emergency, I shall inform the
                    Placement Team in advance by sending a formal email with valid
                    supporting documents to <strong>placements@cs.du.ac.in</strong>.
                  </p>
                  <p>
                    Failure to comply with this protocol may result in my being
                    marked as a defaulter.
                  </p>
                </div>
              </div>
            </div>

            {/* Actions */}
            <div className="flex justify-end gap-4">
              <button
                onClick={() => setShowPopup(false)}
                className="px-6 py-3 bg-gray-200 text-gray-700 rounded-full font-semibold hover:bg-gray-300 transition-all duration-300"
              >
                Cancel
              </button>
              <button
                onClick={handleApply}
                disabled={!isChecked || !resumeLink || isApplying}
                className="bg-[#72265F] text-white px-6 py-3 rounded-full font-semibold hover:bg-[#913e7c] transition-all duration-300 shadow-md hover:shadow-lg transform hover:scale-105 active:scale-95 disabled:bg-gray-400 disabled:cursor-not-allowed disabled:transform-none"
              >
                {isApplying ? "Submitting..." : "Submit Application"}
              </button>
            </div>
          </div>
        </div>
      )}

      <style jsx>{`
        .custom-scrollbar::-webkit-scrollbar {
          width: 8px;
        }
        .custom-scrollbar::-webkit-scrollbar-track {
          background: #f1f1f1;
          border-radius: 4px;
        }
        .custom-scrollbar::-webkit-scrollbar-thumb {
          background: #72265F;
          border-radius: 4px;
        }
        .custom-scrollbar::-webkit-scrollbar-thumb:hover {
          background: #913e7c;
        }
      `}</style>
    </div>
  );
};

export default ViewCompany;
